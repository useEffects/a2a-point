\newcommand{\MyTextField}[3][1]{%
    \noindent
    \makebox[1.8cm][l]{\textbf{#2:}}\quad
    \TextField[name=#3, width=6cm, multiline=true, height=#1\baselineskip, bordercolor={}, borderwidth=0.5pt]{}%
}

}
